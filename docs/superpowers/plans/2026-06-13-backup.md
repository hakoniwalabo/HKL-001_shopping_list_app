# バックアップ・機種変更引継ぎ機能 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** SQLiteデータ（lists・items・favorites）をJSONファイルでエクスポート/インポートし、機種変更時のデータ引継ぎを可能にする

**Architecture:** BackupService が export/import のビジネスロジックを担い、DatabaseHelper.importData() がマージINSERT処理を行う。エクスポートは share_plus の shareXFiles()、インポートは file_picker でファイル選択。インポート時は既存データを残してマージ（同名リストは連番付与）。

**Tech Stack:** Flutter / Dart, sqflite（既存）, share_plus（既存）, file_picker（新規）, path_provider（新規）

```dart
  // -------------------------
  // バックアップ・インポート
  // -------------------------

  /// バックアップJSONデータをマージINSERTする
  Future<ImportResult> importData(Map<String, dynamic> data) async {
    final db = await database;

    final incomingLists =
        List<Map<String, dynamic>>.from(data['lists'] as List);
    final incomingItems =
        List<Map<String, dynamic>>.from(data['items'] as List);
    final incomingFavorites =
        List<Map<String, dynamic>>.from(data['favorites'] as List);

    // 既存リスト名を先に取得（transaction外）
    final existingListRows = await db.query('lists', columns: ['name']);
    final existingNames =
        existingListRows.map((r) => r['name'] as String).toSet();

    final listIdMap = <int, int>{};

    await db.transaction((txn) async {
      // lists INSERT
      for (final list in incomingLists) {
        final oldId = list['id'] as int;
        final resolvedName =
            resolveListName(list['name'] as String, existingNames);
        existingNames.add(resolvedName);

        final newId = await txn.insert('lists', {
          'name': resolvedName,
          'createdAt': list['createdAt'],
        });
        listIdMap[oldId] = newId;
      }

      // items INSERT（listId を新IDに差し替え）
      for (final item in incomingItems) {
        final oldListId = item['listId'] as int;
        final newListId = listIdMap[oldListId];
        if (newListId == null) continue;

        final copy = Map<String, dynamic>.from(item)
          ..remove('id')
          ..['listId'] = newListId;
        await txn.insert('items', copy);
      }

      // favorites INSERT（sortOrder を MAX+1 から採番）
      final maxResult = await txn.rawQuery(
        'SELECT MAX(sortOrder) as maxOrder FROM favorites',
      );
      int nextOrder =
          ((maxResult.first['maxOrder'] as int?) ?? -1) + 1;

      for (final fav in incomingFavorites) {
        final copy = Map<String, dynamic>.from(fav)
          ..remove('id')
          ..['sortOrder'] = nextOrder++;
        await txn.insert('favorites', copy);
      }
    });

    return ImportResult(
      listCount: incomingLists.length,
      itemCount: incomingItems.length,
      favoriteCount: incomingFavorites.length,
    );
  }

  /// リスト名の衝突を解決して一意な名前を返す。
  /// 衝突する場合は末尾に(2)(3)...を付与する。
  static String resolveListName(String name, Set<String> existingNames) {
    if (!existingNames.contains(name)) return name;
    int i = 2;
    while (existingNames.contains('$name($i)')) {
      i++;
    }
    return '$name($i)';
  }
```

- [ ] **Step 3: `lib/services/` ディレクトリと `backup_service.dart` を作成**

```dart
import 'dart:convert';
import 'dart:io';

import 'package:file_picker/file_picker.dart';
import 'package:path_provider/path_provider.dart';
import 'package:share_plus/share_plus.dart';

import '../database/database_helper.dart';

class BackupService {
  /// 全データを JSON ファイルとして共有シートに渡す
  static Future<void> export() async {
    final helper = DatabaseHelper.instance;
    final lists = await helper.getLists();

    final allItems = <Map<String, dynamic>>[];
    for (final list in lists) {
      final items = await helper.getItemsByListId(list.id!);
      allItems.addAll(items.map((i) => i.toMap()));
    }

    final favorites = await helper.getFavorites();

    final data = {
      'version': 1,
      'exportedAt': DateTime.now().millisecondsSinceEpoch,
      'lists': lists.map((l) => l.toMap()).toList(),
      'items': allItems,
      'favorites': favorites,
    };

    final json = const JsonEncoder.withIndent('  ').convert(data);
    final dir = await getTemporaryDirectory();
    final now = DateTime.now();
    final fileName = 'shopping_backup_'
        '${now.year}${_pad(now.month)}${_pad(now.day)}'
        '_${_pad(now.hour)}${_pad(now.minute)}${_pad(now.second)}.json';
    final file = File('${dir.path}/$fileName');
    await file.writeAsString(json);

    await Share.shareXFiles(
      [XFile(file.path)],
      subject: '買い物リスト バックアップ',
    );
  }

  /// ファイルを選択してマージインポート。キャンセル時は null を返す。
  /// FormatException: 'invalid_json' または 'unsupported_version' を投げる可能性あり
  static Future<ImportResult?> import() async {
    final result = await FilePicker.platform.pickFiles(
      type: FileType.custom,
      allowedExtensions: ['json'],
    );

    if (result == null || result.files.isEmpty) return null;
    final path = result.files.first.path;
    if (path == null) return null;

    final content = await File(path).readAsString();
    final data = parseAndValidate(content);
    return await DatabaseHelper.instance.importData(data);
  }

  /// JSON 文字列をパースしてバリデーション済み Map を返す。
  /// 不正な場合は FormatException を投げる。
  static Map<String, dynamic> parseAndValidate(String content) {
    late Map<String, dynamic> data;
    try {
      data = jsonDecode(content) as Map<String, dynamic>;
    } catch (_) {
      throw const FormatException('invalid_json');
    }
    if (data['version'] != 1) {
      throw const FormatException('unsupported_version');
    }
    if (data['lists'] == null ||
        data['items'] == null ||
        data['favorites'] == null) {
      throw const FormatException('invalid_json');
    }
    return data;
  }

  static String _pad(int n) => n.toString().padLeft(2, '0');
}
```

    try {
      await BackupService.export();
    } catch (_) {
      if (!mounted) return;

        const SnackBar(content: Text('バックアップの作成に失敗しました')),

  Future<void> _importBackup() async {
    final confirmed = await showDialog<bool>(
      context: context,
      builder: (_) => AlertDialog(
        title: const Text('バックアップデータを読み込む'),
        content: const Text(
          '既存のデータは残したまま、バックアップのデータが追加されます。よろしいですか？',
        ),
        actions: [
          TextButton(
            onPressed: () => Navigator.of(context).pop(false),
            child: const Text('キャンセル'),
          ),
          FilledButton(
            onPressed: () => Navigator.of(context).pop(true),
            child: const Text('読み込む'),
          ),
        ],
      ),
    );
    if (confirmed != true || !mounted) return;

    final messenger = ScaffoldMessenger.of(context);
    try {
      final result = await BackupService.import();
      if (result == null) return;
      if (!mounted) return;
      await loadLists();
      if (!mounted) return;
      await ref.read(itemProvider.notifier).changeList(currentListId);
      messenger.showSnackBar(
        SnackBar(
          content: Text(
            '${result.listCount}リスト・${result.itemCount}件の商品・'
            '${result.favoriteCount}件のよく買うものを追加しました',
          ),
        ),
      );
    } on FormatException catch (e) {
      messenger.showSnackBar(
        SnackBar(
          content: Text(
            e.message == 'unsupported_version'
                ? '対応していないバックアップ形式です'
                : '読み込めないファイルです',
          ),
        ),
      );
    } catch (_) {
      messenger.showSnackBar(
        const SnackBar(content: Text('読み込みに失敗しました')),
      );
    }
  }
