# カテゴリ機能 実装プラン

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** ユーザーが自由に命名できるカスタムカテゴリを商品・よく買うものに付与し、リスト画面をカテゴリ別セクションに分けて表示する。

**Architecture:** categories テーブル（全リスト共通）を新設し、items・favorites に nullable の categoryId を追加（DB v7）。カテゴリを設定しない商品は「カテゴリなし」セクションに表示され、カテゴリが1件も存在しない場合はヘッダーを省略して現行と同じ表示になる。バックアップ形式は version:2 に更新するが、version:1 の読み込みは引き続きサポートする。

**Tech Stack:** Flutter / Dart、sqflite、flutter_riverpod

```dart
class Item {
  final String name;
  final int price;
  final bool isChecked;
  final int? id;
  final int createdAt;
  final int sortOrder;
  final int quantity;
  final String memo;
  final int listId;
  final int? categoryId;

  Item({
    required this.name,
    required this.price,
    this.isChecked = false,
    this.id,
    int? createdAt,
    this.sortOrder = 0,
    this.quantity = 1,
    this.memo = '',
    required this.listId,
    this.categoryId,
  }) : createdAt = createdAt ?? DateTime.now().millisecondsSinceEpoch;

  Map<String, dynamic> toMap() {
    return {
      'id': id,
      'name': name,
      'price': price,
      'isChecked': isChecked ? 1 : 0,
      'createdAt': createdAt,
      'sortOrder': sortOrder,
      'quantity': quantity,
      'memo': memo,
      'listId': listId,
      'categoryId': categoryId,
    };
  }

  factory Item.fromMap(Map<String, dynamic> map) {
    return Item(
      id: map['id'],
      name: map['name'],
      price: map['price'],
      isChecked: map['isChecked'] == 1,
      createdAt: map['createdAt'],
      sortOrder: map['sortOrder'],
      quantity: map['quantity'] ?? 1,
      memo: map['memo'] ?? '',
      listId: map['listId'],
      categoryId: map['categoryId'] as int?,
    );
  }

  Item copyWith({
    String? name,
    int? price,
    bool? isChecked,
    int? id,
    int? createdAt,
    int? sortOrder,
    int? quantity,
    String? memo,
    int? listId,
    int? categoryId,
  }) {
    return Item(
      name: name ?? this.name,
      price: price ?? this.price,
      isChecked: isChecked ?? this.isChecked,
      id: id ?? this.id,
      createdAt: createdAt ?? this.createdAt,
      sortOrder: sortOrder ?? this.sortOrder,
      quantity: quantity ?? this.quantity,
      memo: memo ?? this.memo,
      listId: listId ?? this.listId,
      categoryId: categoryId ?? this.categoryId,
    );
  }

  /// categoryId を明示的に null にクリアするための専用メソッド。
  /// copyWith は null を「変更なし」と解釈するため使えない。
  Item copyWithNullableCategory({required int? categoryId}) {
    return Item(
      name: name,
      price: price,
      isChecked: isChecked,
      id: id,
      createdAt: createdAt,
      sortOrder: sortOrder,
      quantity: quantity,
      memo: memo,
      listId: listId,
      categoryId: categoryId,
    );
  }
}
```

```dart
Future<List<Map<String, dynamic>>> getCategories() async {
  final db = await database;
  return await db.query('categories', orderBy: 'sortOrder ASC');
}

/// カテゴリを新規作成し、作成した id を返す。
Future<int> insertCategory(String name) async {
  final db = await database;
  final result = await db.rawQuery(
    'SELECT MAX(sortOrder) as maxOrder FROM categories',
  );
  final maxOrder = (result.first['maxOrder'] as int?) ?? -1;
  return await db.insert('categories', {
    'name': name,
    'sortOrder': maxOrder + 1,
    'createdAt': DateTime.now().millisecondsSinceEpoch,
  });
}

/// カテゴリを削除し、紐付く items・favorites の categoryId を NULL にリセットする。
Future<void> deleteCategory(int id) async {
  final db = await database;
  await db.transaction((txn) async {
    await txn.update(
      'items',
      {'categoryId': null},
      where: 'categoryId = ?',
      whereArgs: [id],
    );
    await txn.update(
      'favorites',
      {'categoryId': null},
      where: 'categoryId = ?',
      whereArgs: [id],
    );
    await txn.delete('categories', where: 'id = ?', whereArgs: [id]);
  });
}
```

```dart
// lib/providers/category_provider.dart
import 'package:flutter_riverpod/flutter_riverpod.dart';

import '../database/database_helper.dart';
import '../models/category.dart';

class CategoryNotifier extends AsyncNotifier<List<Category>> {
  @override
  Future<List<Category>> build() async {
    return _fetch();
  }

  Future<List<Category>> _fetch() async {
    final rows = await DatabaseHelper.instance.getCategories();
    return rows.map(Category.fromMap).toList();
  }

  /// カテゴリを新規作成し、作成した id を返す。
  Future<int> addCategory(String name) async {
    final id = await DatabaseHelper.instance.insertCategory(name);
    state = AsyncData(await _fetch());
    return id;
  }

  /// カテゴリを削除する。紐付く商品の categoryId は DB 側で NULL にリセットされる。
  Future<void> deleteCategory(int id) async {
    await DatabaseHelper.instance.deleteCategory(id);
    state = AsyncData(await _fetch());
  }
}

final categoryProvider =
    AsyncNotifierProvider<CategoryNotifier, List<Category>>(
  CategoryNotifier.new,
);
```

```dart
Future<void> showAddItemDialog({
  required BuildContext context,
  required int currentListId,
  required List<String> historyNames,

    context: context,
    builder: (dialogContext) => _AddItemDialogContent(
      currentListId: currentListId,

enum _CloseAction { cancel, discard, add }

class _AddItemDialogContent extends ConsumerStatefulWidget {
  final int currentListId;
  final List<String> historyNames;
  final Item? item;

  const _AddItemDialogContent({
    required this.currentListId,
    required this.historyNames,
    this.item,
  });

  @override
  ConsumerState<_AddItemDialogContent> createState() =>
      _AddItemDialogContentState();
}

class _AddItemDialogContentState extends ConsumerState<_AddItemDialogContent> {
  final _formKey = GlobalKey<FormState>();
  late final TextEditingController _nameCtrl;
  late final TextEditingController _priceCtrl;
  late final TextEditingController _quantityCtrl;
  late final TextEditingController _memoCtrl;

  final FocusNode _nameFocus = FocusNode();
  int _addedCount = 0;
  int? _selectedCategoryId;

  @override
  void initState() {
    super.initState();
    final item = widget.item;
    _nameCtrl = TextEditingController(text: item?.name ?? '');
    _priceCtrl = TextEditingController(text: item?.price.toString() ?? '0');
    _quantityCtrl = TextEditingController(
      text: item?.quantity.toString() ?? '1',
    );
    _memoCtrl = TextEditingController(text: item?.memo ?? '');
    _selectedCategoryId = item?.categoryId;
  }

  @override
  void dispose() {
    _nameCtrl.dispose();
    _priceCtrl.dispose();
    _quantityCtrl.dispose();
    _memoCtrl.dispose();
    _nameFocus.dispose();
    super.dispose();
  }

  Future<void> _submit({bool closeAfter = false}) async {
    if (!_formKey.currentState!.validate()) return;

    final name = _nameCtrl.text.trim();
    final price = int.tryParse(_priceCtrl.text) ?? 0;
    final quantity = int.tryParse(_quantityCtrl.text) ?? 1;

    final item = widget.item;
    if (item != null) {
      final memo = _memoCtrl.text.trim();
      final updatedItem = item
          .copyWith(name: name, price: price, quantity: quantity, memo: memo)
          .copyWithNullableCategory(categoryId: _selectedCategoryId);
      await ref.read(itemProvider.notifier).updateItem(updatedItem);
      if (name != item.name) {
        await DatabaseHelper.instance.upsertHistory(name);
      }
    } else {
      final newItem = Item(
        name: name,
        price: price,
        quantity: quantity,
        memo: '',
        listId: widget.currentListId,
        categoryId: _selectedCategoryId,
      );
      await ref.read(itemProvider.notifier).addItem(newItem);
      await DatabaseHelper.instance.upsertHistory(name);

      if (!mounted) return;
      if (closeAfter) {
        Navigator.of(context).pop();
        return;
      }
      setState(() {
        _addedCount++;
        _selectedCategoryId = _selectedCategoryId; // カテゴリは連続追加でも保持
      });
      _nameCtrl.clear();
      _priceCtrl.text = '0';
      _quantityCtrl.text = '1';
      _nameFocus.requestFocus();
      return;
    }

    if (!mounted) return;
    Navigator.of(context).pop();
  }

  Future<void> _handleClose() async {
    if (_nameCtrl.text.trim().isEmpty) {
      Navigator.of(context).pop();
      return;
    }

    final action = await showDialog<_CloseAction>(
      context: context,
      builder: (ctx) => AlertDialog(
        title: const Text('未追加の入力があります'),
        content: Text('「${_nameCtrl.text.trim()}」はまだ追加されていません。'),
        actionsOverflowDirection: VerticalDirection.down,
        actions: [
          TextButton(
            onPressed: () => Navigator.of(ctx).pop(_CloseAction.cancel),
            child: const Text('入力に戻る'),
          ),
          TextButton(
            onPressed: () => Navigator.of(ctx).pop(_CloseAction.discard),
            child: const Text('破棄して閉じる'),
          ),
          FilledButton(
            onPressed: () => Navigator.of(ctx).pop(_CloseAction.add),
            child: const Text('追加して閉じる'),
          ),
        ],
      ),
    );

    if (!mounted || action == null || action == _CloseAction.cancel) return;
    if (action == _CloseAction.add) {
      await _submit(closeAfter: true);
      return;
    }
    if (!mounted) return;
    Navigator.of(context).pop();
  }

  Future<void> _showNewCategoryDialog(List categories) async {
    String newName = '';
    final confirmed = await showDialog<bool>(
      context: context,
      builder: (ctx) => AlertDialog(
        title: const Text('カテゴリを追加'),
        content: TextField(
          autofocus: true,
          decoration: const InputDecoration(labelText: 'カテゴリ名'),
          onChanged: (v) => newName = v.trim(),
        ),
        actions: [
          TextButton(
            onPressed: () => Navigator.of(ctx).pop(false),
            child: const Text('キャンセル'),
          ),
          FilledButton(
            onPressed: () => Navigator.of(ctx).pop(true),
            child: const Text('追加'),
          ),
        ],
      ),
    );
    if (confirmed != true || newName.isEmpty || !mounted) return;
    final id = await ref.read(categoryProvider.notifier).addCategory(newName);
    setState(() => _selectedCategoryId = id);
  }

  @override
  Widget build(BuildContext context) {
    final isEdit = widget.item != null;
    final categoriesAsync = ref.watch(categoryProvider);
    final categories = categoriesAsync.value ?? [];

    return AlertDialog(
      title: Text(isEdit ? '商品編集' : '商品追加'),
      content: SingleChildScrollView(
        child: Form(
          key: _formKey,
          child: Column(
            mainAxisSize: MainAxisSize.min,
            crossAxisAlignment: CrossAxisAlignment.stretch,
            children: [
              if (!isEdit && _addedCount > 0) ...[
                Text(
                  '$_addedCount件追加しました',
                  style: TextStyle(
                    fontSize: 13,
                    color: Theme.of(context).colorScheme.primary,
                  ),
                ),
                const SizedBox(height: 8),
              ],
              ItemNameField(
                controller: _nameCtrl,
                historyNames: widget.historyNames,
                focusNode: _nameFocus,
              ),
              const SizedBox(height: 12),
              PriceField(controller: _priceCtrl),
              const SizedBox(height: 12),
              QuantityField(controller: _quantityCtrl),
              if (isEdit) ...[
                const SizedBox(height: 12),
                MemoField(controller: _memoCtrl),
              ],
              const SizedBox(height: 12),
              // カテゴリ（任意）
              Row(
                children: [
                  Expanded(
                    child: DropdownButtonFormField<int?>(
                      decoration: const InputDecoration(
                        labelText: 'カテゴリ（任意）',
                        border: OutlineInputBorder(),
                        contentPadding:
                            EdgeInsets.symmetric(horizontal: 12, vertical: 8),
                      ),
                      value: _selectedCategoryId,
                      items: [
                        const DropdownMenuItem<int?>(
                          value: null,
                          child: Text('なし'),
                        ),
                        ...categories.map(
                          (c) => DropdownMenuItem<int?>(
                            value: c.id,
                            child: Text(c.name),
                          ),
                        ),
                      ],
                      onChanged: (v) => setState(() => _selectedCategoryId = v),
                    ),
                  ),
                  const SizedBox(width: 8),
                  IconButton(
                    tooltip: '新規カテゴリ',
                    icon: const Icon(Icons.add_circle_outline),
                    onPressed: () => _showNewCategoryDialog(categories),
                  ),
                ],
              ),
            ],
          ),
        ),
      ),
      actions: [
        TextButton(
          onPressed: isEdit ? () => Navigator.of(context).pop() : _handleClose,
          child: Text(isEdit ? 'キャンセル' : '閉じる'),
        ),
        FilledButton(
          onPressed: _submit,
```

```dart
/// 未購入アイテムをカテゴリ別セクションに分けて返す。
/// カテゴリが1件も存在しない場合はヘッダーなしのフラット表示にする。
List<Widget> _buildCategorySections({
  required List<Item> unchecked,
  required List<Category> categories,
  required NumberFormat numberFormat,
}) {
  // カテゴリが1件もなければ現行どおりフラット表示
  if (categories.isEmpty) {
    return [
      ListSection(
        title: '未購入',
        items: unchecked,
        reorderable: true,
        numberFormat: numberFormat,
        onReorder: (oldIndex, newIndex) async {
          await ref
              .read(itemProvider.notifier)
              .reorderUncheckedItems(oldIndex, newIndex);
        },
        onToggleCheck: (item) async {
          await ref.read(itemProvider.notifier).toggleCheck(item.id!);
        },
        onDelete: _handleDeleteItem,
        onTap: _handleTapItem,
        onMoveToTab:
            shoppingLists.length > 1 ? _handleMoveToTab : null,
      ),
    ];
  }

  final result = <Widget>[];

  // カテゴリごとのセクション
  for (final cat in categories) {
    final catItems =
        unchecked.where((i) => i.categoryId == cat.id).toList();
    if (catItems.isEmpty) continue;
    result.add(
      ListSection(
        title: cat.name,
        items: catItems,
        reorderable: true,
        numberFormat: numberFormat,
        onReorder: (oldIndex, newIndex) async {
          // セクション内のみ並び替え。全体リストでのインデックスに変換
          final globalOld =
              unchecked.indexOf(catItems[oldIndex]);
          final globalNew =
              unchecked.indexOf(catItems[newIndex]);
          await ref
              .read(itemProvider.notifier)
              .reorderUncheckedItems(globalOld, globalNew);
        },
        onToggleCheck: (item) async {
          await ref.read(itemProvider.notifier).toggleCheck(item.id!);
        },
        onDelete: _handleDeleteItem,
        onTap: _handleTapItem,
        onMoveToTab:
            shoppingLists.length > 1 ? _handleMoveToTab : null,
      ),
    );
  }

  // カテゴリなしセクション
  final noCategory =
      unchecked.where((i) => i.categoryId == null).toList();
  if (noCategory.isNotEmpty) {
    result.add(
      ListSection(
        title: 'カテゴリなし',
        items: noCategory,
        reorderable: true,
        numberFormat: numberFormat,
        onReorder: (oldIndex, newIndex) async {
          final globalOld =
              unchecked.indexOf(noCategory[oldIndex]);
          final globalNew =
              unchecked.indexOf(noCategory[newIndex]);
          await ref
              .read(itemProvider.notifier)
              .reorderUncheckedItems(globalOld, globalNew);
        },
        onToggleCheck: (item) async {
          await ref.read(itemProvider.notifier).toggleCheck(item.id!);
        },
        onDelete: _handleDeleteItem,
        onTap: _handleTapItem,
        onMoveToTab:
            shoppingLists.length > 1 ? _handleMoveToTab : null,
      ),
    );
  }

  return result;
}
```

> **注:** `_handleDeleteItem`・`_handleTapItem`・`_handleMoveToTab` は既存メソッドを利用する。SnackBar の Undo 処理など既存の `onDelete` ラムダをそのままメソッドに切り出してリファクタリングすること。

```dart
  final target = lists.where((l) => l.id == listId).toList();
  if (target.isEmpty) return false;

  if (unchecked.isEmpty) return false;

  final allCategories = await helper.getCategories();

      allCategories.where((c) => usedCategoryIds.contains(c['id'])).toList();

  final map = buildExportMap(
    categories: usedCategories,
    lists: [target.first.toMap()],
    items: unchecked.map((i) => i.toMap()).toList(),
    favorites: const [],
    exportedAt: DateTime.now().millisecondsSinceEpoch,
  );
  final json = const JsonEncoder.withIndent('  ').convert(map);

  final now = DateTime.now();
  final fileName = 'shoppinglist_${sanitizeFileName(listName)}_'
      '${now.year}${_pad(now.month)}${_pad(now.day)}.json';
```

```dart
static Map<String, dynamic> parseAndValidate(String content) {
  late Map<String, dynamic> data;
  try {
    data = jsonDecode(content) as Map<String, dynamic>;
  } catch (_) {
    throw const FormatException('invalid_json');
  }
  if (data['version'] is! int) {
    throw const FormatException('invalid_json');
  }
  final version = data['version'] as int;
  if (version != 1 && version != 2) {
    throw const FormatException('unsupported_version');
  }
  if (data['lists'] is! List ||
      data['items'] is! List ||
      data['favorites'] is! List) {
    throw const FormatException('invalid_json');
  }
  if (version == 2 && data['categories'] is! List) {
    throw const FormatException('invalid_json');
  }
  return data;
}
```

```dart
Future<ImportResult> importData(Map<String, dynamic> data) async {
  if (data['lists'] is! List ||
      data['items'] is! List ||
      data['favorites'] is! List) {
    throw ArgumentError('importData: 不正なデータ構造');
  }
  final db = await database;

  final version = data['version'] as int? ?? 1;
  final incomingCategories = version == 2
      ? List<Map<String, dynamic>>.from(data['categories'] as List)
      : <Map<String, dynamic>>[];
  final incomingLists =
      List<Map<String, dynamic>>.from(data['lists'] as List);
  final incomingItems =
      List<Map<String, dynamic>>.from(data['items'] as List);
  final incomingFavorites =
      List<Map<String, dynamic>>.from(data['favorites'] as List);

  final existingListRows = await db.query('lists', columns: ['name']);
  final existingNames =
      existingListRows.map((r) => r['name'] as String).toSet();

  if (existingNames.length + incomingLists.length > 10) {
    throw const FormatException('list_limit_exceeded');
  }

  final listIdMap = <int, int>{};
  final categoryIdMap = <int, int>{};
  int insertedFavCount = 0;

  await db.transaction((txn) async {
    // categories INSERT（name 重複はスキップ）
    final existingCatRows =
        await txn.query('categories', columns: ['name']);
    final existingCatNames =
        existingCatRows.map((r) => r['name'] as String).toSet();
    final maxCatResult = await txn.rawQuery(
      'SELECT MAX(sortOrder) as maxOrder FROM categories',
    );
    int nextCatOrder =
        ((maxCatResult.first['maxOrder'] as int?) ?? -1) + 1;

    for (final cat in incomingCategories) {
      final oldId = (cat['id'] as num).toInt();
      final catName = cat['name'] as String;
      if (existingCatNames.contains(catName)) {
        // 既存カテゴリの ID をマッピング
        final existing = await txn.query(
          'categories',
          columns: ['id'],
          where: 'name = ?',
          whereArgs: [catName],
        );
        if (existing.isNotEmpty) {
          categoryIdMap[oldId] = existing.first['id'] as int;
        }
        continue;
      }
      existingCatNames.add(catName);
      final newId = await txn.insert('categories', {
        'name': catName,
        'sortOrder': nextCatOrder++,
        'createdAt': cat['createdAt'],
      });
      categoryIdMap[oldId] = newId;
    }

    // lists INSERT
    for (final list in incomingLists) {
      final oldId = (list['id'] as num).toInt();
      final resolvedName =
          resolveListName(list['name'] as String, existingNames);
      existingNames.add(resolvedName);

      final newId = await txn.insert('lists', {
        'name': resolvedName,
        'createdAt': list['createdAt'],
      });
      listIdMap[oldId] = newId;
    }

    // items INSERT（listId・categoryId を新IDに差し替え）
    for (final item in incomingItems) {
      final oldListId = (item['listId'] as num).toInt();
      final newListId = listIdMap[oldListId];
      if (newListId == null) continue;

      final oldCategoryId = item['categoryId'] != null
          ? (item['categoryId'] as num).toInt()
          : null;
      final newCategoryId = oldCategoryId != null
          ? categoryIdMap[oldCategoryId]
          : null;

      final copy = Map<String, dynamic>.from(item)
        ..remove('id')
        ..['listId'] = newListId
        ..['categoryId'] = newCategoryId;
      await txn.insert('items', copy);
    }

    // favorites INSERT
    final existingFavRows =
        await txn.query('favorites', columns: ['name']);
    final existingFavNames =
        existingFavRows.map((r) => r['name'] as String).toSet();
    final maxResult = await txn.rawQuery(
      'SELECT MAX(sortOrder) as maxOrder FROM favorites',
    );
    int nextOrder =
        ((maxResult.first['maxOrder'] as int?) ?? -1) + 1;

    for (final fav in incomingFavorites) {
      final favName = fav['name'] as String;
      if (existingFavNames.contains(favName)) continue;
      existingFavNames.add(favName);

      final oldCategoryId = fav['categoryId'] != null
          ? (fav['categoryId'] as num).toInt()
          : null;
      final newCategoryId = oldCategoryId != null
          ? categoryIdMap[oldCategoryId]
          : null;

      final copy = Map<String, dynamic>.from(fav)
        ..remove('id')
        ..['sortOrder'] = nextOrder++
        ..['categoryId'] = newCategoryId;
      await txn.insert('favorites', copy);
      insertedFavCount++;
    }
  });

  return ImportResult(
    listCount: incomingLists.length,
    itemCount: incomingItems.length,
    favoriteCount: insertedFavCount,
  );
}
```
